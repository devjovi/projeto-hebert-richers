class Ingresso:
    def __init__(self, codigo: str, nome_evento: str, valor_base: float):
        self.codigo = codigo
        self.nome_evento = nome_evento
        self.valor_base = float(valor_base)

    def calcular_valor_final(self) -> float:
        return round(self.valor_base, 2)


class IngressoPadrao(Ingresso):
    def calcular_valor_final(self) -> float:
        return round(self.valor_base, 2)


class IngressoVIP(Ingresso):
    def __init__(
        self,
        codigo: str,
        nome_evento: str,
        valor_base: float,
        beneficios: list[str] | None = None,
    ):
        super().__init__(codigo, nome_evento, valor_base)
        self.beneficios = (
            beneficios
            if beneficios is not None
            else ["Acesso VIP", "Atendimento prioritário"]
        )

    def calcular_valor_final(self) -> float:
        return round(self.valor_base * 1.20, 2)
